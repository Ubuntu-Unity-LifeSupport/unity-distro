---- mixrace.py (verifier 2)
#!/usr/bin/python3
# Verifier-2 UNITY-20260928-022: N calls in flight for D seconds, cycling over
# every method and property of /org/gnome/SettingsDaemon/Power; counts reply
# kinds and reports the longest reply time. Usage: mixrace.py N D
import sys, time, collections, itertools
import gi
gi.require_version("Gio", "2.0")
from gi.repository import Gio, GLib
N, D = int(sys.argv[1]), float(sys.argv[2])
PW="org.gnome.SettingsDaemon.Power"
CALLS = [
 (PW+".Keyboard","StepUp",None),(PW+".Keyboard","StepDown",None),(PW+".Keyboard","Toggle",None),
 (PW+".Screen","GetPercentage",None),(PW+".Screen","SetPercentage",GLib.Variant("(u)",(50,))),
 (PW+".Screen","StepUp",None),(PW+".Screen","StepDown",None),
 (PW,"GetDevices",None),(PW,"GetPrimaryDevice",None),
 ("org.freedesktop.DBus.Properties","Get",GLib.Variant("(ss)",(PW,"Percentage"))),
 ("org.freedesktop.DBus.Properties","Get",GLib.Variant("(ss)",(PW,"Icon"))),
 ("org.freedesktop.DBus.Properties","Get",GLib.Variant("(ss)",(PW,"Tooltip"))),
 ("org.freedesktop.DBus.Properties","GetAll",GLib.Variant("(s)",(PW,))),
 ("org.freedesktop.DBus.Introspectable","Introspect",None),
]
cyc = itertools.cycle(CALLS)
bus = Gio.bus_get_sync(Gio.BusType.SESSION, None)
loop = GLib.MainLoop()
counts = collections.Counter(); maxt=[0.0]
end = time.monotonic() + D
inflight = [0]
def done(conn, res, data):
    key, t0 = data
    inflight[0] -= 1
    dt = time.monotonic()-t0; maxt[0]=max(maxt[0],dt)
    try:
        conn.call_finish(res); counts[key+":reply"] += 1
    except GLib.Error as e:
        m = e.message
        for k in ("not running","No keyboard backlight","UnknownMethod","NoReply","Timeout","timed out",
                  "ServiceUnknown","NameHasNoOwner","Disconnected","not available","no primary device","Screen backlight"):
            if k in m:
                counts[key+":"+k] += 1
                if k in ("Timeout","timed out","NoReply","Disconnected"): print("  %s %s at %s" % (k,key,time.strftime("%T")), flush=True)
                break
        else: counts[key+":other:"+m[:60]] += 1
    fire()
def fire():
    if time.monotonic() < end:
        iface, meth, args = next(cyc)
        key = iface.split(".")[-1]+"."+meth+("("+args.unpack()[-1]+")" if meth=="Get" else "")
        inflight[0] += 1
        bus.call(PW, "/org/gnome/SettingsDaemon/Power", iface, meth, args, None,
                 Gio.DBusCallFlags.NONE, 10000, None, done, (key, time.monotonic()))
    elif inflight[0] == 0:
        loop.quit()
for _ in range(N): fire()
loop.run()
for k in sorted(counts): print("   %-60s %d" % (k, counts[k]))
print("  max reply time: %.3f s" % maxt[0])

#!/usr/bin/python3
# Verifier UNITY-20260928-022: keep N Keyboard.StepUp calls in flight to the
# power object for D seconds, count reply kinds. Usage: kbdrace.py N D [method]
import sys, time, collections
import gi
gi.require_version("Gio", "2.0")
from gi.repository import Gio, GLib
N, D = int(sys.argv[1]), float(sys.argv[2])
IFACE, METHOD = (sys.argv[3].rsplit(".", 1) if len(sys.argv) > 3
                 else ("org.gnome.SettingsDaemon.Power.Keyboard", "StepUp"))
bus = Gio.bus_get_sync(Gio.BusType.SESSION, None)
loop = GLib.MainLoop()
counts = collections.Counter()
end = time.monotonic() + D
inflight = [0]
def done(conn, res, _):
    inflight[0] -= 1
    try:
        conn.call_finish(res); counts["reply"] += 1
    except GLib.Error as e:
        m = e.message
        for k in ("not running", "UnknownMethod", "NoReply", "Timeout", "timed out",
                  "ServiceUnknown", "NameHasNoOwner", "Disconnected"):
            if k in m:
                counts[k] += 1
                if k in ("Timeout","timed out","NoReply") and counts[k] in (1,): print("  first %s at %s" % (k, time.strftime("%T")), flush=True)
                break
        else: counts["other:" + m[:80]] += 1
    fire()
def fire():
    if time.monotonic() < end:
        inflight[0] += 1
        bus.call("org.gnome.SettingsDaemon.Power", "/org/gnome/SettingsDaemon/Power",
                 IFACE, METHOD, None, None, Gio.DBusCallFlags.NONE, 10000, None, done, None)
    elif inflight[0] == 0:
        loop.quit()
for _ in range(N): fire()
loop.run()
print("  calls by result:", dict(counts))
